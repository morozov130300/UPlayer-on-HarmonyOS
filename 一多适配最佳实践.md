#### 2.1.2 Tabs布局设备适配

> 1. 在手机及折叠屏设备上，tabs菜单位于屏幕底部（tabs采用垂直布局，barPosition属性为End）
> 2. 在平板设备上，tabs菜单位于屏幕的左侧（tabs采用水平布局，barPosition属性为Start）

![image-20250528013359650](imgs/image-20250528013359650.png)

**技术点：使用屏幕的尺寸断点**

`横向断点`以应用窗口宽度为判断条件，建议划分为5个区间：

| 断点名称 | 窗口宽度（vp） | 代表设备             |
| :------- | :------------- | -------------------- |
| xs       | (0, 320）      | 手表watch            |
| sm       | [320, 600)     | 常规手机             |
| md       | [600, 840)     | 双折叠展开、平板竖屏 |
| lg       | [840, 1440)    | 三折叠展开、平板横屏 |
| xl       | [1440, +∞)     | pc                   |

`纵向断点`根据应用窗口的**高宽比**进行判断，建议划分为3个区间：

| 断点名称 | 高宽比     |
| :------- | :--------- |
| sm       | (0, 0.8)   |
| md       | [0.8, 1.2) |
| lg       | [1.2, +∞)  |

   ```typescript
   getWindowBreakpoint(windowStage: window.WindowStage){
       //获取当前设备屏幕的断点（xs、sm、md、lg、xl）
       windowStage.getMainWindow().then((data:window.Window)=>{
           let uiConext = data.getUIContext()
           //WidthBreakpoint.WIDTH_XS、WIDTH_SM、WIDTH_MD、WIDTH_LG、WIDTH_XL
           let widthBreakPoint:WidthBreakpoint = uiConext.getWindowWidthBreakpoint()
           //HeightBreakpoint.HEIGHT_SM 、 HEIGHT_MD 、HEIGHT_LG
           let heightBreakPoint:HeightBreakpoint = uiConext.getWindowHeightBreakpoint()
           //将获取到的横向断点和纵向断点存储到AppStorage
           AppStorage.setOrCreate('widthBreakPoint',widthBreakPoint)
           AppStorage.setOrCreate('heightBreakPoint',heightBreakPoint)
           //订阅屏幕尺寸变化事件
           data.on('windowSizeChange',()=>{
               let widthBreakPoint:WidthBreakpoint = uiConext.getWindowWidthBreakpoint()
               //HeightBreakpoint.HEIGHT_SM 、 HEIGHT_MD 、HEIGHT_LG
               let heightBreakPoint:HeightBreakpoint = uiConext.getWindowHeightBreakpoint()
               //将获取到的横向断点和纵向断点存储到AppStorage
               AppStorage.setOrCreate('widthBreakPoint',widthBreakPoint)
               AppStorage.setOrCreate('heightBreakPoint',heightBreakPoint)
           })
       })
   }
   ```

   

2. 定义断点取值帮助类，以根据不同的断点获取组件对应属性值/参数

   - 在ets中新建名为`utils`的directory

   - 在`utils`下新建名为`BreakpointType`的ArkTS文件

   - 定义屏幕断点值如下：

     ````typescript
     export class WidthBreakpointType<T>{
         xs:T
         sm:T
         md:T
         lg:T
         xl:T
     
         constructor(xs: T, sm: T, md: T, lg: T, xl: T) {
             this.xs = xs
             this.sm = sm
             this.md = md
             this.lg = lg
             this.xl = xl
         }
     
         getValue(widthBreakpoint:WidthBreakpoint):T{
             if(widthBreakpoint === WidthBreakpoint.WIDTH_XS){
                 return this.xs
             }
             if(widthBreakpoint === WidthBreakpoint.WIDTH_SM){
                 return this.sm
             }
             if(widthBreakpoint === WidthBreakpoint.WIDTH_MD){
                 return this.md
             }
             if(widthBreakpoint === WidthBreakpoint.WIDTH_LG){
                 return this.lg
             }
             if(widthBreakpoint === WidthBreakpoint.WIDTH_XL){
                 return this.xl
             }
             return this.sm
         }
     
     }
     ````

getCoursesByWeek() {
    if (this.rdbStore) {
        let predicates: relationalStore.RdbPredicates =
            new relationalStore.RdbPredicates("course")
        predicates.equalTo('school_year',this.currentSchoolYear)
        predicates.equalTo('school_term',this.currentSchoolTerm)
        predicates.lessThanOrEqualTo("start_week",this.currentWeek)
        predicates.greaterThanOrEqualTo("end_week",this.currentWeek)
        // 查询结果按星期升序，再按节升序
        predicates.orderByAsc("weekday").orderByAsc("start_section")
        this.rdbStore.query(predicates).then(resultSet => {
            //resultSet就是查询结果集：将resultSet数据取出，存储到自己的容器中
            let result:Array<Course> = []
            while (resultSet.goToNextRow()) {
                let course = new Course()
                course.id = resultSet.getString(resultSet.getColumnIndex('id'))
                course.name = resultSet.getString(resultSet.getColumnIndex('name'))
                course.teacher = resultSet.getString(resultSet.getColumnIndex('teacher'))
                course.location = resultSet.getString(resultSet.getColumnIndex('location'))
                course.weekday = resultSet.getLong(resultSet.getColumnIndex('weekday'))
                course.startSection = 
                    resultSet.getLong(resultSet.getColumnIndex('start_section'))
                course.endSection = resultSet.getLong(resultSet.getColumnIndex('end_section'))
                course.startWeek = resultSet.getLong(resultSet.getColumnIndex('start_week'))
                course.endWeek = resultSet.getLong(resultSet.getColumnIndex('end_week'))
                course.weekType = resultSet.getString(resultSet.getColumnIndex('week_type'))
                course.schoolYear = 
                    resultSet.getString(resultSet.getColumnIndex('school_year'))
                course.schoolTerm = 
                    resultSet.getString(resultSet.getColumnIndex('school_term'))
                course.note = resultSet.getString(resultSet.getColumnIndex('note'))
                course.colorTag = resultSet.getString(resultSet.getColumnIndex('color_tag'))
                result.push(course)
            }
            this.courseList = result

                console.log(course.name,course.location,course.weekday);
            })
            // 释放数据集的内存

getRealtimeWeatherByCityId(cityId:string){
    let httpRequest = http.createHttp()
    let url = `https://mx57rnakx5.re.qweatherapi.com/v7/weather/now?location=${cityId}`
    let options:http.HttpRequestOptions = {
        method: http.RequestMethod.GET,
        header: {
            'X-QW-Api-Key': '83288c679ee04c4d90b624b3c7e65471'  //使用请求天气信息的api_key
        }
    }
    httpRequest.request(encodeURI(url),options).then((response)=>{
        if(response.responseCode == 200){

            let weatherNowResp = JSON.parse(jsonStr) as WeatherNowResp
